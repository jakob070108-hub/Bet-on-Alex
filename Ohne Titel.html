        // Lokale Datenbank initialisieren

        if(!localStorage.getItem('boa_users')) localStorage.setItem('boa_users', JSON.stringify({}));

        if(!localStorage.getItem('boa_bets')) localStorage.setItem('boa_bets', JSON.stringify([]));

        if(!localStorage.getItem('boa_placements')) localStorage.setItem('boa_placements', JSON.stringify([]));


        let currentUser = null;


        function loadData(key) { return JSON.parse(localStorage.getItem(key)); }

        function saveData(key, data) { localStorage.setItem(key, JSON.stringify(data)); }

        function hash(str) { return btoa(encodeURIComponent(str)); }


        function login() {

            const user = document.getElementById('login-username').value.trim();

            const pass = document.getElementById('login-password').value.trim();

            const errorEl = document.getElementById('login-error');


            if(!user || !pass) {

                errorEl.innerText = "Bitte Benutzername und Passwort eingeben!";

                errorEl.classList.remove('hidden');

                return;

            }


            let users = loadData('boa_users');


            if(!users[user]) {

                users[user] = {

                    password: hash(pass),

                    role: user.toLowerCase() === 'admin' ? 'admin' : 'user',

                    status: user.toLowerCase() === 'admin' ? 'verified' : 'pending',

                    balance: 0

                };

                saveData('boa_users', users);

            } else {

                if(users[user].password !== hash(pass)) {

                    errorEl.innerText = "Falsches Passwort!";

                    errorEl.classList.remove('hidden');

                    return;

                }

            }


            errorEl.classList.add('hidden');

            currentUser = user;

            updateUI();

        }


        function logout() {

            currentUser = null;

            document.getElementById('login-username').value = '';

            document.getElementById('login-password').value = '';

            updateUI();

        }


        function updateUI() {

            document.getElementById('login-view').classList.add('hidden');

                document.getElementById('login-view').classList.remove('hidden');

                return;

            }


            const users = loadData('boa_users');

            const userData = users[currentUser];


            document.getElementById('header').classList.remove('hidden');

            

            if(userData.role === 'admin') {

                document.getElementById('user-info').innerText = `Admin-Modus`;

                document.getElementById('admin-view').classList.remove('hidden');

                renderAdminView();

            } else if (userData.status === 'pending') {

                document.getElementById('user-info').innerText = `Eingeloggt als ${currentUser}`;

                document.getElementById('pending-view').classList.remove('hidden');

            } else {

                document.getElementById('user-info').innerText = `${currentUser} | Guthaben: ${userData.balance.toFixed(2)} Coins`;

                document.getElementById('user-view').classList.remove('hidden');

        function verifyUser(username) {

            let users = loadData('boa_users');

            let balance = parseFloat(document.getElementById(`start-${username}`).value);

            if(isNaN(balance) || balance < 0) return alert("Bitte ein gültiges Startguthaben eingeben.");

            

            users[username].status = 'verified';

            users[username].balance = balance;

            saveData('boa_users', users);

            renderAdminView();

        }

        function createBet() {

            const title = document.getElementById('new-bet-title').value.trim();

            if(!title) return alert("Bitte ein Wett-Ereignis eintragen.");


            const rows = document.querySelectorAll('.bet-option-row');

            let options = [];

            

            rows.forEach(row => {

                const text = row.children[0].value.trim();

                const odds = parseFloat(row.children[1].value);

                if(text && !isNaN(odds) && odds > 1) {

                    options.push({ text: text, odds: odds });

                }

            });


            if(options.length < 2) return alert("Eine Wette benötigt mindestens 2 gültige Optionen mit Quoten über 1.0.");


            let bets = loadData('boa_bets');

            bets.push({

                title: title,

                options: options,

                status: 'open',

                winnerId: null

            });

            saveData('boa_bets', bets);

            

            document.getElementById('new-bet-title').value = '';

            document.getElementById('bet-options-container').innerHTML = `

                <div class="flex gap-2 bet-option-row">

                    <input type="text" placeholder="Option" class="w-2/3 p-2.5 rounded-lg bg-slate-900 border border-slate-700 text-white focus:outline-none">

                    <input type="number" step="0.01" placeholder="Quote" class="w-1/3 p-2.5 rounded-lg bg-slate-900 border border-slate-700 text-white focus:outline-none">

                </div>

                <div class="flex gap-2 bet-option-row mt-2">

                    <input type="text" placeholder="Option" class="w-2/3 p-2.5 rounded-lg bg-slate-900 border border-slate-700 text-white focus:outline-none">

                    <input type="number" step="0.01" placeholder="Quote" class="w-1/3 p-2.5 rounded-lg bg-slate-900 border border-slate-700 text-white focus:outline-none">

                </div>`;

            renderAdminView();

            alert("Wette erfolgreich erstellt!");

        }


        function resolveBet(betId) {

            const selectEl = document.getElementById(`winner-${betId}`);

            if(selectEl.value === "") return alert("Bitte wähle ein Gewinner-Ergebnis aus.");

            const winnerIdx = parseInt(selectEl.value);


            let bets = loadData('boa_bets');

            let users = loadData('boa_users');

            let placements = loadData('boa_placements');


            bets[betId].status = 'closed';

            bets[betId].winnerId = winnerIdx;


            const winningOdds = bets[betId].options[winnerIdx].odds;


            placements.forEach(p => {

                if(p.betId === betId && p.status === 'pending') {

                    if(p.optionIdx === winnerIdx) {

                        p.status = 'won';

                        users[p.username].balance += p.amount * winningOdds;

                    } else {

                        p.status = 'lost';

                    }

                }

            });


            saveData('boa_bets', bets);

            saveData('boa_users', users);

            saveData('boa_placements', placements);

            renderAdminView();

            alert("Wette erfolgreich ausgewertet!");

        }

                }

            });

            if(!hasOpenBets) activeBetsContainer.innerHTML = '<p class="text-sm text-slate-400">Derzeit sind keine Wetten verfügbar.</p>';


            const myPlacements = placements.filter(p => p.username === currentUser).reverse();

            if(myPlacements.length === 0) historyContainer.innerHTML = '<p class="text-sm text-slate-400">Du hast noch keine Wetten platziert.</p>';


            myPlacements.forEach(p => {

                const bet = bets[p.betId];

                if (!bet) return;

                const option = bet.options[p.optionIdx];

                let statusBadge = '<span class="text-amber-400 font-semibold">Offen</span>';

                if(p.status === 'won') statusBadge = `<span class="text-emerald-400 font-bold">Gewonnen (+${(p.amount * option.odds).toFixed(2)})</span>`;

                if(p.status === 'lost') statusBadge = '<span class="text-rose-400 font-semibold">Verloren</span>';


                historyContainer.innerHTML += `

                    <div class="flex justify-between items-center bg-slate-900 p-3 rounded-xl border border-slate-700 text-sm">

                        <div>

                            <p class="font-bold text-slate-200">${bet.title}</p>

                            <p class="text-slate-400 text-xs mt-0.5">Tipp: ${option.text} (Quote: ${option.odds})</p>

                        </div>

                        <div class="text-right">

                            <p class="text-slate-300 font-medium">${p.amount} Coins</p>

                            <p class="mt-0.5">${statusBadge}</p>

                        </div>

                    </div>

                `;

            });

        }


        function placeBet(betId, optionIdx) {

            const amountInput = document.getElementById(`bet-amt-${betId}-${optionIdx}`);

            const amount = parseFloat(amountInput.value);

            

            if(isNaN(amount) || amount <= 0) return alert("Bitte einen gültigen Einsatz eingeben.");


            let users = loadData('boa_users');

            if(users[currentUser].balance < amount) return alert("Nicht genügend Guthaben auf dem Konto!");


            users[currentUser].balance -= amount;


            let placements = loadData('boa_placements');

            placements.push({

                username: currentUser,

                betId: betId,

                optionIdx: optionIdx,

                amount: amount,

                status: 'pending'

            });


            saveData('boa_users', users);

            saveData('boa_placements', placements);

            

            alert("Wette erfolgreich platziert!");

            updateUI();

        }


        // Backup Funktionen für Admin

        function exportData() {

            const data = {

                users: loadData('boa_users'),

                bets: loadData('boa_bets'),

                placements: loadData('boa_placements')

            };

            const blob = new Blob([JSON.stringify(data, null, 2)], { type: 'json' });

            const url = URL.createObjectURL(blob);

            const a = document.createElement('a');

            a.href = url;

            a.download = `bet-on-alex-backup-${new Date().toISOString().slice(0,10)}.json`;

            a.click();

        }


        function importData(event) {

            const file = event.target.files[0];

            if(!file) return;

            const reader = new FileReader();

            reader.onload = function(e) {

                try {

                    const data = JSON.parse(e.target.result);

                    if(data.users && data.bets) {

                        saveData('boa_users', data.users);

                        saveData('boa_bets', data.bets);

                        saveData('boa_placements', data.placements || []);

                        alert("Daten erfolgreich importiert!");

                        updateUI();

                    } else {

                        alert("Ungültige Backup-Datei.");

                    }

                } catch(err) {

                    alert("Fehler beim Lesen der Datei.");

                }

            };

            reader.readAsText(file);

        }

    </script>
